// ========== EXISTING NAVIGATION CODE ==========
const navLinks = document.querySelectorAll('.sidebar-nav li');
const contentSections = document.querySelectorAll('.tab-content');
const dashboardMainContent = document.getElementById('dashboard-section');

navLinks.forEach(link => {
    link.addEventListener('click', function(e) {
        e.preventDefault();
        
        const targetTab = this.getAttribute('data-tab');
        
        // Hide all tab contents
        contentSections.forEach(section => {
            section.classList.remove('active');
        });
        
        // Show the selected section
        const targetSection = document.getElementById(`${targetTab}-section`);
        if (targetSection) {
            targetSection.classList.add('active');
        }
        
        // Update active state on navigation
        navLinks.forEach(l => l.classList.remove('active'));
        this.classList.add('active');
    });
});


const dashboardLink = document.querySelector('.sidebar-nav li[data-tab="dashboard"]');

if (dashboardLink) {
    dashboardLink.addEventListener('click', function(e) {
        e.preventDefault();

        // Remove active from all sections
        contentSections.forEach(section => {
            section.classList.remove('active');
        });

        // Show dashboard ONLY
        if (dashboardMainContent) {
            dashboardMainContent.classList.add('active');
        }

        // Update sidebar active state
        navLinks.forEach(l => l.classList.remove('active'));
        this.classList.add('active');
    });
}
